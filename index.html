@startuml

title SAGHIRI - Class Diagram

skinparam packageStyle rectangle

hide empty members

' ==========================================
' 1. BOUNDARY CLASS
' ==========================================

package "Boundary Layer" {

    class MobileAppUI <> {
        +register()
        +login()
        +logout()
        +manageProfile()
        +manageChildInformation()
        +recordBabyCry()
        +viewAnalysisResult()
        +sendChatMessage()
        +viewVitalSigns()
        +viewHistory()
        +viewCharts()
        +viewAlerts()
    }
}

' ==========================================
' 2. CONTROL CLASSES
' ==========================================

package "Control Layer" {

    class AccountController <> {
        +register()
        +login()
        +logout()
        +changePassword()
        +resetPassword()
    }

    class ProfileController <> {
        +viewProfile()
        +editProfile()
    }

    class ChildController <> {
        +viewChildInformation()
        +editChildInformation()
    }

    class CryAnalysisController <> {
        +recordCry()
        +sendRecordingForAnalysis()
        +receiveAnalysisResult()
        +displayAnalysisResult()
    }

    class ChatbotController <> {
        +sendMessage()
        +receiveResponse()
    }

    class HealthMonitoringController <> {
        +receiveSensorData()
        +checkAbnormalValues()
        +saveVitalSigns()
        +viewMeasurements()
        +viewHistory()
        +viewCharts()
        +generateAlerts()
        +viewAlerts()
    }

    class VitalSignLocalStorage <> {
        +saveVitalSign(vitalSign)
        +getVitalSigns()
        +getVitalSignsByDate(date)
    }
}

' ==========================================
' 3. ENTITY CLASSES
' ==========================================

package "Entity Layer" {

    class Parent <> {
        -name: String
        -email: String
        -passwordHash: String
    }

    class Child <> {
        -firstName: String
        -lastName: String
        -dateOfBirth: Date
        -gender: Gender
    }

    class VitalSign <> {
        -heartRate: Float
        -oxygenSaturation: Float
        -temperature: Float
        -recordedAt: DateTime
    }

    class HealthAlert <> {
        -message: String
        -createdAt: DateTime
    }

    class CryRecording <> {
        -audioFile: File
        -recordedAt: DateTime
    }

    class CryAnalysisResult <> {
        -result: String
        -analyzedAt: DateTime
    }

    class ChatMessage <> {
        -content: String
        -sender: String
        -timestamp: DateTime
    }

    enum Gender
}

' ==========================================
' 4. EXTERNAL SYSTEMS
' ==========================================

package "External Systems" {

    class ESP32 <> {
        +sendSensorData()
    }

    class AIService <> {
        +analyzeCry()
    }
}

' ==========================================
' 5. BOUNDARY - CONTROL RELATIONSHIPS
' ==========================================

MobileAppUI ..> AccountController : uses
MobileAppUI ..> ProfileController : uses
MobileAppUI ..> ChildController : uses
MobileAppUI ..> CryAnalysisController : uses
MobileAppUI ..> ChatbotController : uses
MobileAppUI ..> HealthMonitoringController : uses

' ==========================================
' 6. CONTROL - ENTITY RELATIONSHIPS
' ==========================================

AccountController ..> Parent : manages
ProfileController ..> Parent : manages
ChildController ..> Child : manages

CryAnalysisController ..> CryRecording : handles
CryAnalysisController ..> CryAnalysisResult : handles

ChatbotController ..> ChatMessage : handles

HealthMonitoringController ..> VitalSign : handles
HealthMonitoringController ..> HealthAlert : generates

' ==========================================
' 7. LOCAL HEALTH DATA STORAGE
' ==========================================

HealthMonitoringController ..> VitalSignLocalStorage : uses

VitalSignLocalStorage ..> VitalSign : saves / retrieves

' ==========================================
' 8. EXTERNAL SYSTEM INTERACTIONS
' ==========================================

ESP32 ..> HealthMonitoringController : sends sensor data

CryAnalysisController ..> AIService : sends audio recording

AIService ..> CryAnalysisController : returns analysis result

' ==========================================
' 9. ENTITY ASSOCIATIONS
' ==========================================

Parent "1" -- "1" Child : has

Child "1" -- "0..*" VitalSign : has readings

Child "1" -- "0..*" CryRecording : records

CryRecording "1" -- "0..1" CryAnalysisResult : produces

VitalSign "1" -- "0..*" HealthAlert : may trigger

' ==========================================
' 10. NOTES
' ==========================================

note right of VitalSign
  Each reading contains all three values:
  - Heart Rate
  - Oxygen Saturation
  - Temperature
end note

note right of CryRecording
  Temporary recording.
  Sent for analysis without
  permanent storage.
end note

note right of CryAnalysisResult
  Displayed by the application.
  Permanent storage is not confirmed.
end note

note right of ChatMessage
  Temporary messages.
  Chat history is not stored.
end note

note bottom of VitalSignLocalStorage
  Local storage inside the mobile phone.
  Used to retrieve historical readings
  and generate History charts.
end note

Login Sequence Diagram

Show Diagram Code 

@startuml
actor Parent
boundary "MobileAppUI" as UI
control "AccountController" as AC
participant "AccountRepository" as AR
database "PostgreSQL" as DB

Parent -> UI: enterEmailAndPassword(email, password)
activate UI

Parent -> UI: clickLogin()
UI -> AC: login(email, password)
activate AC

AC -> AR: findByEmail(email)
activate AR

AR -> DB: retrieveAccount(email)
activate DB
DB --> AR: accountData
deactivate DB

AR --> AC: accountData
deactivate AR

AC -> AC: verifyPassword(password, passwordHash)

alt [credentials valid]
    AC --> UI: authenticationSuccess()
    UI --> Parent: displayHomeScreen()
else [credentials invalid or account not found]
    AC --> UI: authenticationFailure()
    UI --> Parent: displayErrorMessage()
end

deactivate AC
deactivate UI
@enduml

ChatInterface -> ChatController : sendMessage(text)
activate ChatController
create entity ChatMessage
ChatController -> ChatMessage : <> (sender = Parent)
ChatController -> Child : getAge()
activate Child
Child --> ChatController : age (computed by Flutter)
deactivate Child
ChatController -> ChatbotProxy : sendMessage(text, childName, childAge)
activate ChatbotProxy
ChatbotProxy -> ChatAI : question + name + age
activate ChatAI
alt reply received
  ChatAI --> ChatbotProxy : general advice
  ChatbotProxy --> ChatController : reply
  ChatController -> ChatMessage : <> (sender = Assistant)
  ChatController --> ChatInterface : reply
  ChatInterface --> Parent : show reply
else service failure
  ChatAI --> ChatbotProxy : error
  deactivate ChatAI
  ChatbotProxy --> ChatController : error
  deactivate ChatbotProxy
  ChatController --> ChatInterface : error
  ChatInterface --> Parent : show error message
end
deactivate ChatController

activate MonitoringController

MonitoringInterface --> Parent : show current values
deactivate MonitoringController

== View Monitoring Charts ==
Parent -> MonitoringInterface : choose History / time range
MonitoringInterface -> MonitoringController : buildChartData(range)
activate MonitoringController
MonitoringController -> LocalStorageProxy : loadReadings(range)
activate LocalStorageProxy
LocalStorageProxy --> MonitoringController : stored readings
deactivate LocalStorageProxy
alt readings available
  MonitoringController --> MonitoringInterface : chart data
  MonitoringInterface --> Parent : show charts
else no readings
  MonitoringController --> MonitoringInterface : empty
  MonitoringInterface --> Parent : "No data available"
end
